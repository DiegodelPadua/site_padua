"use client";

import { useEffect, useRef, useState } from "react";

/* =========================================================

type UsuarioCheckout = {
  id: number;
  nome: string;
  sobrenome: string;
  email: string;
  telefone?: string | null;
};

type EnderecoCheckout = {
  id: number;
  apelido: string | null;
  destinatario: string;
  cep: string;
  logradouro: string;
  numero: string;
  complemento: string | null;
  bairro: string;
  cidade: string;
  estado: string;
  principal: boolean;
};

type DadosEntrega = {
  cep: string;
  endereco: string;
  numero: string;
  complemento: string;
  bairro: string;
  cidade: string;
  estado: string;
};

type OpcaoFreteCheckout = {
  id: number;
  nome: string;
  transportadora: string;
  preco: number;
  prazoDias: number | null;
};

const entregaInicial: DadosEntrega = {
  cep: "",
  endereco: "",
  numero: "",
  complemento: "",
  bairro: "",
  cidade: "",
  estado: "",
};

function formatarMoeda(valor: number) {
  return valor.toLocaleString("pt-BR", {
    style: "currency",
    currency: "BRL",
  });
}

function somenteNumeros(valor: string) {
  return valor.replace(/\D/g, "");
}

export default function Checkout() {
  /* =======================================================
     SACOLA
     ======================================================= */

  const { items } = useCart();

  const subtotal = items.reduce(
    (total, item) => total + item.price * item.quantity,
    0
  );

  const quantidadeItens = items.reduce(
    (total, item) => total + item.quantity,
    0
  );

  /* =======================================================
     AUTENTICAÃ‡ÃƒO
     ======================================================= */

  const [usuario, setUsuario] =
    useState<UsuarioCheckout | null>(null);

  const [carregandoUsuario, setCarregandoUsuario] =
    useState(true);

  const [erroUsuario, setErroUsuario] = useState("");

  useEffect(() => {
    const controller = new AbortController();

    async function consultarCliente() {
      try {
        const resposta = await fetch("/api/auth/me", {
          method: "GET",
          credentials: "same-origin",
          cache: "no-store",
          signal: controller.signal,
        });

        if (resposta.status === 401) {
          setUsuario(null);
          return;
        }

        if (!resposta.ok) {
          throw new Error(
            "NÃ£o foi possÃ­vel consultar sua conta."
          );
        }

        const dados = await resposta.json();

        if (!controller.signal.aborted) {
          setUsuario(
            dados.autenticado && dados.usuario
              ? dados.usuario
              : null
          );
        }
      } catch {
        if (!controller.signal.aborted) {
          setErroUsuario(
            "NÃ£o foi possÃ­vel carregar seus dados."
          );
        }
      } finally {
        if (!controller.signal.aborted) {
          setCarregandoUsuario(false);
        }
      }
    }

    void consultarCliente();

    return () => controller.abort();
  }, []);

  /* =======================================================
     ENDEREÃ‡OS
     ======================================================= */

  const [enderecos, setEnderecos] =
    useState<EnderecoCheckout[]>([]);

  const [
    enderecoSelecionadoId,
    setEnderecoSelecionadoId,
  ] = useState<number | null>(null);

  const [
    carregandoEnderecos,
    setCarregandoEnderecos,
  ] = useState(true);

  const [salvandoEndereco, setSalvandoEndereco] =
    useState(false);

  const [mensagemEndereco, setMensagemEndereco] =
    useState("");

  const [dadosEntrega, setDadosEntrega] =
    useState<DadosEntrega>(entregaInicial);

  // Impede respostas antigas de sobrescreverem
  // informaÃ§Ãµes digitadas manualmente.
  const versaoEdicaoEndereco = useRef(0);

  const consultaCepController =
    useRef<AbortController | null>(null);

  const [consultandoCep, setConsultandoCep] =
    useState(false);

  const [mensagemCep, setMensagemCep] = useState("");

  /* =======================================================
     FRETE
     ======================================================= */

  const [opcoesFrete, setOpcoesFrete] =
    useState<OpcaoFreteCheckout[]>([]);

  const [freteSelecionadoId, setFreteSelecionadoId] =
    useState<number | null>(null);

  const [cotandoFrete, setCotandoFrete] =
    useState(false);

  const [erroFrete, setErroFrete] = useState("");

  const [mensagemPagamento, setMensagemPagamento] =
    useState("");

  const cotacaoController =
    useRef<AbortController | null>(null);

  const cepCotacao = somenteNumeros(dadosEntrega.cep);

  // Detecta alteraÃ§Ãµes na sacola, inclusive quantidade.
  const chaveSacola = JSON.stringify(
    items.map((item) => [
      item.id,
      item.variantId,
      item.quantity,
    ])
  );

  // quando o CEP ou os itens mudam.
  useEffect(() => {
    cotacaoController.current?.abort();
    cotacaoController.current = null;

    setOpcoesFrete([]);
    setFreteSelecionadoId(null);
    setErroFrete("");
    setMensagemPagamento("");
    setCotandoFrete(false);
  }, [cepCotacao, chaveSacola]);

  /* =======================================================
     FUNÃ‡Ã•ES DE ENDEREÃ‡O
     ======================================================= */

    versaoEdicaoEndereco.current += 1;

    consultaCepController.current?.abort();
    consultaCepController.current = null;

    setConsultandoCep(false);
    setMensagemCep("");

    setEnderecoSelecionadoId(null);
    setMensagemEndereco("");

    consultaCepController.current?.abort();
    consultaCepController.current = null;

    versaoEdicaoEndereco.current += 1;

    setConsultandoCep(false);
    setMensagemCep("");

    setDadosEntrega({
      cep: endereco.cep,
      endereco: endereco.logradouro,
      numero: endereco.numero,
      complemento: endereco.complemento ?? "",
      bairro: endereco.bairro,
      cidade: endereco.cidade,
      estado: endereco.estado,
    });
  }

    campo: keyof DadosEntrega,
    valor: string
  ) {
    registrarEdicaoEndereco();

    setDadosEntrega((anterior) => ({
      ...anterior,
      [campo]: valor,

      setMensagemEndereco("");

        const resposta = await fetch("/api/enderecos", {
          method: "GET",
          credentials: "same-origin",
          cache: "no-store",

          );

          Array.isArray(dados.enderecos)
            ? dados.enderecos

        setEnderecos(lista);

          lista[0];

          principal &&
          versaoEdicaoEndereco.current === versaoInicial

        if (!controller.signal.aborted) {
          setCarregandoEnderecos(false);
        }
      }

    return () => controller.abort();
  }, [usuario?.id, carregandoUsuario]);

  /* =======================================================
     CONSULTA AUTOMÃTICA DE CEP â€” VIACEP
     ======================================================= */

  async function consultarCEP(
    cep: string,
    versaoDaDigitacao: number
  ) {
    const numeros = somenteNumeros(cep);

    if (numeros.length !== 8) return;

    consultaCepController.current?.abort();

    const controller = new AbortController();
    consultaCepController.current = controller;

    setConsultandoCep(true);
    setMensagemCep("");

    try {
      const resposta = await fetch(
        `https://viacep.com.br/ws/${numeros}/json/`,
        { signal: controller.signal }
      );

      if (!resposta.ok) {
        throw new Error(
          "ServiÃ§o de CEP indisponÃ­vel. Preencha manualmente."
        );
      }

      const dados: {
        erro?: boolean;
        logradouro?: string;
        bairro?: string;
        localidade?: string;
        uf?: string;
      } = await resposta.json();

      if (
        controller.signal.aborted ||
        versaoEdicaoEndereco.current !==
          versaoDaDigitacao
      ) {
        return;
      }

      if (dados.erro) {
        setMensagemCep(
          "CEP nÃ£o encontrado. Confira o nÃºmero informado."
        );
        return;
      }

      // NÃºmero e complemento sÃ£o preservados.
      setDadosEntrega((anterior) => ({
        ...anterior,
        endereco: dados.logradouro ?? "",
        bairro: dados.bairro ?? "",
        cidade: dados.localidade ?? "",
        estado: dados.uf ?? "",
      }));

      if (!dados.logradouro || !dados.bairro) {
        setMensagemCep(
          "CEP localizado. Complete os campos nÃ£o informados."
        );
      }
    } catch (erro) {
      if (
        !controller.signal.aborted &&
        versaoEdicaoEndereco.current ===
          versaoDaDigitacao
      ) {
        setMensagemCep(
          erro instanceof Error
            ? erro.message
            : "NÃ£o foi possÃ­vel consultar o CEP."
        );
      }
    } finally {
      if (consultaCepController.current === controller) {
        consultaCepController.current = null;
        setConsultandoCep(false);
      }
    }
  }

  function alterarCEP(valor: string) {
    const numeros = somenteNumeros(valor).slice(0, 8);

    const cepFormatado =
      numeros.length > 5
        ? `${numeros.slice(0, 5)}-${numeros.slice(5)}`
        : numeros;

    registrarEdicaoEndereco();

    const versaoAtual =
      versaoEdicaoEndereco.current;

    // Limpa os dados de um CEP anterior.
    setDadosEntrega((anterior) => ({
      ...anterior,
      cep: cepFormatado,
      endereco: "",
      bairro: "",
      cidade: "",
      estado: "",
    }));

    if (numeros.length === 8) {
      void consultarCEP(numeros, versaoAtual);
    }
  }

  async function salvarEndereco() {
    if (salvandoEndereco) return;

    if (!usuario) {
      setMensagemEndereco(
        "Entre na sua conta para salvar o endereÃ§o."

    if (
      !/^\d{8}$/.test(cepCotacao) ||

      !dadosEntrega.numero.trim() ||
      !dadosEntrega.bairro.trim() ||
      !dadosEntrega.cidade.trim() ||
      !/^[A-Z]{2}$/.test(dadosEntrega.estado.trim())
    ) {
      setMensagemEndereco(
        "Preencha corretamente todos os campos obrigatÃ³rios."
      );
      return;

    setSalvandoEndereco(true);
    setMensagemEndereco("");

    try {
      const resposta = await fetch("/api/enderecos", {
        method: "POST",
        credentials: "same-origin",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          destinatario:
            `${usuario.nome} ${usuario.sobrenome}`.trim(),
          cep: dadosEntrega.cep,
          logradouro: dadosEntrega.endereco,
          numero: dadosEntrega.numero,
          complemento: dadosEntrega.complemento,
          bairro: dadosEntrega.bairro,
          cidade: dadosEntrega.cidade,
          estado: dadosEntrega.estado,
        }),
      });

      const resultado = await resposta.json();

      if (!resposta.ok) {
        throw new Error(
          resultado.erro ??
            "NÃ£o foi possÃ­vel salvar o endereÃ§o."
        );
      }

      const enderecoSalvo =
        resultado.endereco as EnderecoCheckout;

      setEnderecos((anteriores) => [
        enderecoSalvo,
        ...anteriores,
      ]);

      preencherEndereco(enderecoSalvo);

      setMensagemEndereco(
        "EndereÃ§o salvo com sucesso."
      );
    } catch (erro) {
      setMensagemEndereco(
        erro instanceof Error
          ? erro.message
          : "NÃ£o foi possÃ­vel salvar o endereÃ§o."
      );
    } finally {
      setSalvandoEndereco(false);
    }
  }

     CALCULAR FRETE â€” API MELHOR ENVIO

  async function calcularFrete() {
    setErroFrete("");
    setMensagemPagamento("");
    setFreteSelecionadoId(null);
    setOpcoesFrete([]);

    if (!usuario) {
      setErroFrete(
        "Entre na sua conta para calcular o frete."
      );
      return;
    }

    if (!/^\d{8}$/.test(cepCotacao)) {
      setErroFrete(
        "Informe um CEP de entrega vÃ¡lido."
      );
      return;
    }

    if (items.length === 0) {
      setErroFrete("Sua sacola estÃ¡ vazia.");
      return;
    }

    // O ID textual do carrinho nÃ£o substitui
    // o ID numÃ©rico da variante no banco.
    const itemSemVariante = items.some(
      (item) =>
        !Number.isSafeInteger(item.variantId) ||
        (item.variantId ?? 0) <= 0
    );

    if (itemSemVariante) {
      setErroFrete(
        "Existem produtos antigos na sacola. Remova-os e adicione novamente pela loja."
      );
      return;
    }

    cotacaoController.current?.abort();

    const controller = new AbortController();
    cotacaoController.current = controller;

    setCotandoFrete(true);

    try {
      const resposta = await fetch(
        "/api/frete/cotacao",
        {
          method: "POST",
          credentials: "same-origin",
          cache: "no-store",
          headers: {
            "Content-Type": "application/json",
          },
          signal: controller.signal,
          body: JSON.stringify({
            cepDestino: cepCotacao,
            itens: items.map((item) => ({
              variantId: item.variantId,
              quantity: item.quantity,
            })),
          }),
        }
      );

      const dados = await resposta.json();

      if (!resposta.ok) {
        throw new Error(
          dados.erro ??
            "NÃ£o foi possÃ­vel calcular o frete."
        );
      }

      const opcoes: OpcaoFreteCheckout[] =
        Array.isArray(dados.opcoes)
          ? dados.opcoes.filter(
              (opcao: OpcaoFreteCheckout) =>
                Number.isSafeInteger(opcao.id) &&
                Number.isFinite(opcao.preco) &&
                opcao.preco >= 0
            )
          : [];

      if (controller.signal.aborted) return;

      setOpcoesFrete(opcoes);

      if (opcoes.length === 0) {
        setErroFrete(
          "Nenhuma modalidade de entrega disponÃ­vel para este CEP."
        );
      }
    } catch (erro) {
      if (!controller.signal.aborted) {
        setErroFrete(
          erro instanceof Error
            ? erro.message
            : "Falha ao consultar o frete."
        );
      }
    } finally {
      if (cotacaoController.current === controller) {
        cotacaoController.current = null;
        setCotandoFrete(false);
      }
    }
  }

  const freteSelecionado = opcoesFrete.find(
    (opcao) => opcao.id === freteSelecionadoId
  );

  const total =
    subtotal + (freteSelecionado?.preco ?? 0);

export function validarChaveIdempotencia(

/* =========================================================
   VALIDAR DADOS DO NOVO PEDIDO
   ========================================================= */

export function validarDadosNovoPedido(
  entrada: unknown
): DadosNovoPedido {
  if (
    !entrada ||
    typeof entrada !== "object" ||
    Array.isArray(entrada)
  ) {
    throw new PedidoServiceError(
      "Dados do pedido invÃ¡lidos."
    );
  }

  const dados = entrada as Record<string, unknown>;

  /* =====================================================
     1. VALIDAR ENDEREÃ‡O
     ===================================================== */

  if (!inteiroPositivo(dados.enderecoId)) {
    throw new PedidoServiceError(
      "Selecione um endereÃ§o de entrega vÃ¡lido."
    );
  }

  /* =====================================================
     2. VALIDAR FRETE
     =====================================================
     O identificador serÃ¡ conferido novamente com
     uma cotaÃ§Ã£o oficial realizada no servidor.
     ===================================================== */

  if (!inteiroPositivo(dados.freteId)) {
    throw new PedidoServiceError(
      "Selecione uma modalidade de frete vÃ¡lida."
    );
  }

  /* =====================================================
     3. VALIDAR ITENS
     ===================================================== */

  if (
    !Array.isArray(dados.itens) ||
    dados.itens.length === 0 ||
    dados.itens.length > 30
  ) {
    throw new PedidoServiceError(
      "O pedido deve conter entre 1 e 30 itens."
    );
  }

  // Agrupa variantes repetidas na mesma compra.
  const itensAgrupados = new Map<number, number>();

  for (const item of dados.itens) {
    if (
      !item ||
      typeof item !== "object" ||
      Array.isArray(item)
    ) {
      throw new PedidoServiceError(
        "Um dos itens do pedido Ã© invÃ¡lido."
      );
    }

    const atual = item as Record<string, unknown>;

    if (
      !inteiroPositivo(atual.variantId) ||
      !inteiroPositivo(atual.quantity) ||
      atual.quantity > 100
    ) {
      throw new PedidoServiceError(
        "Produto ou quantidade invÃ¡lida."
      );
    }

    const quantidadeAnterior =
      itensAgrupados.get(atual.variantId) ?? 0;

    const quantidadeTotal =
      quantidadeAnterior + atual.quantity;

    if (quantidadeTotal > 100) {
      throw new PedidoServiceError(
        "Quantidade mÃ¡xima excedida para um produto."
      );
    }

    itensAgrupados.set(
      atual.variantId,
      quantidadeTotal
    );
  }

  /* =====================================================
     4. RETORNAR DADOS NORMALIZADOS
     ===================================================== */

  return {
    enderecoId: dados.enderecoId,
    freteId: dados.freteId,
    itens: Array.from(
      itensAgrupados,
      ([variantId, quantity]) => ({
        variantId,
        quantity,
      })
    ),
  };
}

export async function prepararNovoPedido(
  usuarioId: number,
  entrada: unknown
) {
  /* =====================================================
     1. VALIDAR CLIENTE E DADOS RECEBIDOS
     ===================================================== */

  if (
    !Number.isSafeInteger(usuarioId) ||

  /* =====================================================

  const endereco = await buscarEnderecoDoCliente(

  if (!endereco) {
    throw new PedidoServiceError(
      "EndereÃ§o nÃ£o encontrado para este cliente.",
      404

  /* =====================================================
     3. CONSULTAR PRODUTOS ATIVOS
     ===================================================== */

  const variantes = await buscarVariantesDoPedido(
    dados.itens.map((item) => item.variantId)
  );

  if (variantes.length !== dados.itens.length) {
    throw new PedidoServiceError(
      "Um ou mais produtos nÃ£o estÃ£o disponÃ­veis.",
      409
    );
  }

  // Facilita a localizaÃ§Ã£o de cada variante pelo ID.
  const variantesPorId = new Map(
    variantes.map((variante) => [
      variante.id,
      variante,
    ])
  );

  /* =====================================================
     4. VERIFICAR ESTOQUE E CALCULAR SUBTOTAL
     =====================================================
     Prisma.Decimal evita imprecisÃµes nos cÃ¡lculos
     monetÃ¡rios.
     ===================================================== */

  let subtotal = new Prisma.Decimal(0);

  const itens = dados.itens.map((item) => {
    const variante = variantesPorId.get(item.variantId);

    if (!variante) {
      throw new PedidoServiceError(
        "Produto nÃ£o encontrado.",
        409
      );
    }

    // Confere o estoque disponÃ­vel no momento da consulta.
    if (variante.quantidade < item.quantity) {
      throw new PedidoServiceError(
        `Estoque insuficiente para ${variante.produto.nome} (${variante.cor}, ${variante.tamanho}).`,
        409
      );
    }

    // Aplica o preÃ§o promocional quando estiver definido.
    const preco =
      variante.produto.precoPromocional ??

    if (preco.lessThan(0)) {
      throw new PedidoServiceError(
        "PreÃ§o invÃ¡lido para um dos produtos.",
        409
      );
    }

    const subtotalItem = preco.mul(item.quantity);

    subtotal = subtotal.plus(subtotalItem);

    // Preserva os dados do produto para o pedido.
    return {
      varianteId: variante.id,
      nomeProduto: variante.produto.nome,
      tamanho: variante.tamanho,
      cor: variante.cor,
      quantidade: item.quantity,
      precoUnitario: preco,
      subtotal: subtotalItem,
      pesoGramas: variante.produto.pesoGramas,
      categoriaSlug: variante.produto.categoria.slug,
    };
  });

  /* =====================================================
     5. RETORNAR DADOS PREPARADOS
     =====================================================
     O frete ainda serÃ¡ validado na prÃ³xima funÃ§Ã£o.
     ===================================================== */

  return {
    usuarioId,

    freteId: dados.freteId,

  };
}

   Ordenamos os produtos para que a mesma compra

  (a, b) => a.variantId - b.variantId
);

  usuarioId,
  enderecoId: dadosValidados.enderecoId,

  itens: itensOrdenados.map((item) => ({
    variantId: item.variantId,
    quantity: item.quantity,
  })),
};

// Calcula o SHA-256 dos dados da tentativa de checkout.

  .update(JSON.stringify(dadosParaHash))
  .digest("hex");

  (entrada as Record<string, unknown>).chaveIdempotencia

  // Impede que um cliente acesse pedidos de outro usuÃ¡rio.
  if (pedidoExistente.usuarioId !== usuarioId) {

      409
    );
  }

  // Retorna a compra original sem reservar estoque novamente.
  // O status e o prazo da reserva continuam sendo os originais.
  return {

    status: pedidoExistente.status,

  // Tenta registrar o pedido e reservar o estoque.

// ServiÃ§o responsÃ¡vel pela criaÃ§Ã£o do pedido.

import { NextRequest, NextResponse } from "next/server";

import { prisma } from "../../lib/prisma";
import { validarAuthToken } from "../utils/authToken";

/* =========================================================
   CONTROLLER - CRIAR PEDIDO
   ========================================================= */

export async function criarPedidoController(
  request: NextRequest
) {
  try {
    /* =====================================================
       1. IDENTIFICAR O COOKIE DE AUTENTICAÃ‡ÃƒO
       ===================================================== */

    const token = request.cookies.get("padua_session")?.value;

    if (!token) {
      return NextResponse.json(
        { erro: "UsuÃ¡rio nÃ£o autenticado." },
        { status: 401 }
      );
    }

    /* =====================================================
       2. VALIDAR O JWT
       ===================================================== */

    let usuarioId: number;
    let versaoSessao: number;

    try {
      const payload = await validarAuthToken(token);

      usuarioId = Number(payload.usuarioId);
      versaoSessao = Number(payload.versaoSessao);
    } catch {
      return NextResponse.json(
        { erro: "SessÃ£o invÃ¡lida ou expirada." },
        { status: 401 }
      );
    }

    /* =====================================================
       3. VERIFICAR SE A SESSÃƒO FOI REVOGADA
       ===================================================== */

    const usuario = await prisma.usuario.findUnique({
      where: {
        id: usuarioId,
      },
      select: {
        id: true,
        versaoSessao: true,
      },
    });

    if (
      !usuario ||
      usuario.versaoSessao !== versaoSessao
    ) {
      return NextResponse.json(
        { erro: "SessÃ£o revogada. FaÃ§a login novamente." },
        { status: 401 }
      );
    }

    /* =====================================================
       4. VALIDAR A ORIGEM DA REQUISIÃ‡ÃƒO
       =====================================================
       ProteÃ§Ã£o contra requisiÃ§Ãµes externas que tentem
       utilizar o cookie de autenticaÃ§Ã£o do cliente.
       ===================================================== */

    const origin = request.headers.get("origin");

    if (!origin) {
      return NextResponse.json(
        { erro: "Origem da requisiÃ§Ã£o nÃ£o informada." },
        { status: 403 }
      );
    }

    try {
      const origem = new URL(origin);
      const destino = new URL(request.url);

      if (origem.origin !== destino.origin) {
        return NextResponse.json(
          { erro: "Origem da requisiÃ§Ã£o nÃ£o autorizada." },
          { status: 403 }
        );
      }
    } catch {
      return NextResponse.json(
        { erro: "Origem da requisiÃ§Ã£o invÃ¡lida." },
        { status: 403 }
      );
    }

    /* =====================================================
       5. VALIDAR O CORPO DA REQUISIÃ‡ÃƒO
       ===================================================== */

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { erro: "JSON invÃ¡lido." },
        { status: 400 }
      );
    }

    if (
      !body ||
      typeof body !== "object" ||
      Array.isArray(body)
    ) {
      return NextResponse.json(
        { erro: "Dados do pedido invÃ¡lidos." },
        { status: 400 }
      );
    }

    /* =====================================================
       6. PREPARAR A CRIAÃ‡ÃƒO DO PEDIDO
       =====================================================
       O Service serÃ¡ implementado na prÃ³xima etapa.

       Ainda nÃ£o criamos pedidos nem movimentamos estoque.
       ===================================================== */

      // Cria o pedido com os dados validados pelo Service.
      // O Service calcula os preÃ§os, valida o frete e
      // registra a reserva de estoque no MySQL.
      const pedido = await criarNovoPedido(usuarioId, body);

      return NextResponse.json(
        {
          mensagem: "Pedido registrado com sucesso.",
          pedido: {
            id: pedido.id,
            status: pedido.status,
            valorTotal: pedido.valorTotal.toString(),
            criadoEm: pedido.criadoEm,
          },
        },
        {
          status: 201,
          headers: {
            "Cache-Control": "no-store",
          },
        }
      );

  } catch (error) {
    // Erros esperados das regras de negÃ³cio.
    if (error instanceof PedidoServiceError) {
      return NextResponse.json(
        { erro: error.message },
        { status: error.status }
      );
    }

    // Falta de estoque identificada na transaÃ§Ã£o.
    if (error instanceof ReservaEstoqueError) {
      return NextResponse.json(
        { erro: error.message },
        { status: 409 }
      );
    }
    console.error(
      "[PEDIDO CONTROLLER] Erro ao processar pedido:",
      error
    );

    return NextResponse.json(
      { erro: "NÃ£o foi possÃ­vel processar o pedido." },
      { status: 500 }
    );
  }
