export async function registrarPedido(
  dados: DadosRegistroPedido
) {
  return prisma.$transaction(async (tx) => {
    /* =====================================================
       1. CRIAR O PEDIDO PRINCIPAL
       ===================================================== */

    const pedido = await tx.pedido.create({
      data: {
        usuarioId: dados.usuarioId,
        // Chave Ãºnica da tentativa de compra.
        chaveIdempotencia: dados.chaveIdempotencia,

        // Hash que identifica os dados da solicitaÃ§Ã£o original.
        hashCheckout: dados.hashCheckout,

        // Prazo mÃ¡ximo da reserva de estoque.
        reservaExpiraEm: dados.reservaExpiraEm,
        status: StatusPedido.AGUARDANDO_PAGAMENTO,
        subtotal: dados.subtotal,
        valorFrete: dados.valorFrete,
        valorTotal: dados.valorTotal,

      },
      select: {
        id: true,
        status: true,
        valorTotal: true,
        criadoEm: true,
      },
    });


    /* =====================================================
   RESERVAR ESTOQUE DOS PRODUTOS
   =====================================================
   A atualizaÃ§Ã£o condicional garante que somente
   produtos com estoque suficiente sejam reservados.

   Toda a operaÃ§Ã£o ocorre dentro da transaÃ§Ã£o.
   ===================================================== */

for (const item of dados.itens) {
  // Confere se a quantidade solicitada Ã© vÃ¡lida.
  if (
    !Number.isSafeInteger(item.quantidade) ||
    item.quantidade <= 0
  ) {
    throw new ReservaEstoqueError(
      "Quantidade invÃ¡lida para reserva."
    );
  }

  // LÃª o estoque atual da variante.
  const variante = await tx.varianteProduto.findUnique({
    where: {
      id: item.varianteId,
    },
    select: {
      quantidade: true,
    },
  });

  if (!variante) {
    throw new ReservaEstoqueError(
      `Variante ${item.varianteId} nÃ£o encontrada.`
    );
  }

  // Reduz o estoque somente quando hÃ¡ unidades suficientes.
  const resultado = await tx.varianteProduto.updateMany({
    where: {
      id: item.varianteId,
      ativa: true,
      quantidade: {
        gte: item.quantidade,
      },
      produto: {
        ativo: true,
        categoria: {
          ativa: true,
        },
      },
    },
    data: {
      quantidade: {
        decrement: item.quantidade,
      },
    },
  });

  // Se nenhuma linha foi atualizada, a reserva falhou.
  if (resultado.count !== 1) {
    throw new ReservaEstoqueError(
      `Estoque insuficiente para ${item.nomeProduto} (${item.cor}, ${item.tamanho}).`
    );
  }

  // Consulta o saldo efetivamente gravado apÃ³s a reserva.
  const varianteAtualizada =
    await tx.varianteProduto.findUniqueOrThrow({
      where: {
        id: item.varianteId,
      },
      select: {
        quantidade: true,
      },
    });

      const saldoPosterior = varianteAtualizada.quantidade;

      // ReconstrÃ³i o saldo anterior a partir da baixa efetiva.
      // NÃ£o usa a leitura anterior, que pode estar desatualizada

/**
 * Libera o estoque de um pedido cuja reserva expirou.
 *
 * Regras:
 * - Somente pedidos AGUARDANDO_PAGAMENTO podem ser cancelados.
 * - A reserva precisa estar vencida.
 * - Pagamentos pendentes, processando ou aprovados bloqueiam
 *   a liberaÃ§Ã£o atÃ© que sua situaÃ§Ã£o seja conciliada.
 * - O cancelamento e a devoluÃ§Ã£o ocorrem na mesma transaÃ§Ã£o.
 *
 * Retorna true quando a reserva foi liberada.
 * Retorna false quando nÃ£o Ã© seguro ou necessÃ¡rio liberÃ¡-la.
 */
export async function liberarReservaExpirada(
  pedidoId: number
): Promise<boolean> {
  return prisma.$transaction(async (tx) => {
    const agora = new Date();

    // Bloqueia logicamente o pedido por meio de uma atualizaÃ§Ã£o
    // condicional. Apenas uma execuÃ§Ã£o poderÃ¡ mudar seu status.
    const alteracao = await tx.pedido.updateMany({
      where: {
        id: pedidoId,
        status: StatusPedido.AGUARDANDO_PAGAMENTO,
        reservaExpiraEm: {
          lte: agora,
        },

        // NÃ£o libera estoque enquanto houver uma cobranÃ§a
        // que possa ser confirmada pelo provedor.
        pagamentos: {
          none: {
            status: {
              in: [
                StatusPagamento.PENDENTE,
                StatusPagamento.PROCESSANDO,
                StatusPagamento.APROVADO,
              ],
            },
          },
        },
      },
      data: {
        status: StatusPedido.CANCELADO,
      },
    });

    // Se outra execuÃ§Ã£o jÃ¡ alterou o pedido ou existe
    // pagamento que exige conciliaÃ§Ã£o, nÃ£o faz nada.
    if (alteracao.count !== 1) {
      return false;
    }

    // Recupera as reservas efetivamente registradas.
    const reservas = await tx.movimentacaoEstoque.findMany({
      where: {
        pedidoId,
        tipo: TipoMovimentacaoEstoque.RESERVA,
      },
      select: {
        varianteId: true,
        quantidade: true,
      },
    });

    // Devolve cada quantidade reservada ao estoque.
    for (const reserva of reservas) {
      const variante = await tx.varianteProduto.findUnique({
        where: {
          id: reserva.varianteId,
        },
        select: {
          quantidade: true,
        },
      });

      if (!variante) {
        // Qualquer erro desfaz toda a transaÃ§Ã£o,
        // inclusive a mudanÃ§a de status do pedido.
        throw new Error(
          `Variante ${reserva.varianteId} nÃ£o encontrada.`
        );
      }

      // Incremento atÃ´mico para preservar alteraÃ§Ãµes concorrentes.
      await tx.varianteProduto.update({
        where: {
          id: reserva.varianteId,
        },
        data: {
          quantidade: {
            increment: reserva.quantidade,
          },
        },
      });

      // Consulta o saldo apÃ³s a devoluÃ§Ã£o.
      const varianteAtualizada =
        await tx.varianteProduto.findUniqueOrThrow({
          where: {
            id: reserva.varianteId,
          },
          select: {
            quantidade: true,
          },
        });

      // Registra a devoluÃ§Ã£o para auditoria do estoque.
      await tx.movimentacaoEstoque.create({
        data: {
          pedidoId,
          varianteId: reserva.varianteId,
          tipo: TipoMovimentacaoEstoque.LIBERACAO_RESERVA,
          quantidade: reserva.quantidade,
          motivo: "Reserva expirada sem pagamento confirmado.",
          saldoAnterior: varianteAtualizada.quantidade - reserva.quantidade,
          saldoPosterior: varianteAtualizada.quantidade,
        },
      });
    }

    // Registra o cancelamento no histÃ³rico do pedido.
    await tx.historicoPedido.create({
      data: {
        pedidoId,
        status: StatusPedido.CANCELADO,
        descricao: "Pedido cancelado por expiraÃ§Ã£o da reserva de estoque.",
      },
    });

    return true;
  });

export function validarDadosNovoPedido(
  entrada: unknown
): DadosNovoPedido {
  if (

  }

    );

  // Agrupa variantes repetidas na mesma compra.
  const itensAgrupados = new Map<number, number>();

  for (const item of dados.itens) {

      throw new PedidoServiceError(
        "Um dos itens do pedido Ã© invÃ¡lido."

    const atual = item as Record<string, unknown>;

    if (

    const quantidadeAnterior =

    if (quantidadeTotal > 100) {
      throw new PedidoServiceError(

    itensAgrupados.set(

  return {

    const subtotalItem = preco.mul(item.quantity);

    subtotal = subtotal.plus(subtotalItem);

    // Preserva os dados do produto para o pedido.
    return {
      varianteId: variante.id,
      nomeProduto: variante.produto.nome,

  if (

    // Consulta novamente o pedido que venceu a concorrÃªncia.
    const pedidoExistente =
      await buscarPedidoPorChaveIdempotencia(chaveIdempotencia);

    // Se nÃ£o existir pedido com essa chave, o conflito
    // pode ter ocorrido em outra restriÃ§Ã£o UNIQUE.
    if (!pedidoExistente) {
      throw erro;
    }

    // Impede acesso a pedidos de outro cliente.
    if (pedidoExistente.usuarioId !== usuarioId) {
      throw new PedidoServiceError(
        "Chave de idempotÃªncia jÃ¡ utilizada.",
        409
      );
    }

    // A chave nÃ£o pode representar duas compras diferentes.
    if (
      !pedidoExistente.hashCheckout ||
      pedidoExistente.hashCheckout !== hashCheckout
    ) {
      throw new PedidoServiceError(
        "Esta chave jÃ¡ foi utilizada com dados diferentes.",
        409
      );
    }

    // Retorna o pedido original sem reservar estoque novamente.
    return {
      id: pedidoExistente.id,
      status: pedidoExistente.status,
      valorTotal: pedidoExistente.valorTotal,
      criadoEm: pedidoExistente.criadoEm,
    };
  }

  // Erros de estoque, banco ou outros problemas
  // nÃ£o devem ser tratados como pedidos duplicados.
  throw erro;
}

/* =========================================================
   MERCADO PAGO SERVICE - PÃDUA CULTURE
   =========================================================
   IntegraÃ§Ã£o com Checkout Pro pela API de Orders.

   Este serviÃ§o:
   - Recebe dados de um pedido jÃ¡ validado.
   - Cria uma order no Mercado Pago.
   - Retorna o ID e a URL de pagamento.

   NÃ£o deve ser chamado diretamente pelo navegador.
   ========================================================= */

import { Prisma } from "@prisma/client";

const MERCADO_PAGO_API =
  "https://api.mercadopago.com/v1/orders";

/* =========================================================
   TIPOS DE DADOS
   ========================================================= */

export type ItemPagamentoMercadoPago = {
  nome: string;
  quantidade: number;
  precoUnitario: Prisma.Decimal;
};

export type DadosPagamentoMercadoPago = {
  pedidoId: number;
  valorTotal: Prisma.Decimal;
  itens: ItemPagamentoMercadoPago[];

  // Chave persistida e reutilizada nas tentativas
  // de repetiÃ§Ã£o da mesma operaÃ§Ã£o.
  chaveIdempotencia: string;
};

type RespostaOrderMercadoPago = {
  id?: string;
  checkout_url?: string;
  status?: string;
};

/* =========================================================
   CRIAR ORDER DE PAGAMENTO
   ========================================================= */

export async function criarOrderMercadoPago(
  dados: DadosPagamentoMercadoPago
) {
  /* =====================================================
     1. VALIDAR CONFIGURAÃ‡ÃƒO
     ===================================================== */

  const accessToken =
    process.env.MERCADO_PAGO_ACCESS_TOKEN;

  if (!accessToken) {
    throw new Error(
      "Access Token do Mercado Pago nÃ£o configurado."
    );
  }

  /* =====================================================
     2. VALIDAR IDENTIFICADORES E VALORES
     ===================================================== */

  if (
    !Number.isSafeInteger(dados.pedidoId) ||
    dados.pedidoId <= 0
  ) {
    throw new Error("Identificador do pedido invÃ¡lido.");
  }

  if (
    !dados.valorTotal.isFinite() ||
    dados.valorTotal.lessThanOrEqualTo(0)
  ) {
    throw new Error("Valor do pedido invÃ¡lido.");
  }

  if (
    !Array.isArray(dados.itens) ||
    dados.itens.length === 0
  ) {
    throw new Error("Pedido sem itens para pagamento.");
  }

  // UUID vÃ¡lido, gerado e persistido pelo backend.
  const formatoUUID =
    /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  if (!formatoUUID.test(dados.chaveIdempotencia)) {
    throw new Error("Chave de idempotÃªncia invÃ¡lida.");
  }

  /* =====================================================
     3. PREPARAR OS ITENS
     ===================================================== */

  const itens = dados.itens.map((item) => {
    if (
      !item.nome.trim() ||
      !Number.isSafeInteger(item.quantidade) ||
      item.quantidade <= 0 ||
      !item.precoUnitario.isFinite() ||
      item.precoUnitario.lessThan(0)
    ) {
      throw new Error("Item de pagamento invÃ¡lido.");
    }

    return {
      title: item.nome,
      quantity: item.quantidade,
      unit_price: item.precoUnitario.toFixed(2),
    };
  });

  /* =====================================================
     4. CONFERIR A SOMA DOS ITENS
     =====================================================
     O Mercado Pago exige que a soma dos itens
     corresponda ao total da order.
     ===================================================== */

  const somaItens = dados.itens.reduce(
    (total, item) =>
      total.plus(
        item.precoUnitario.mul(item.quantidade)
      ),
    new Prisma.Decimal(0)
  );

  if (!somaItens.equals(dados.valorTotal)) {
    throw new Error(
      "A soma dos itens nÃ£o corresponde ao valor total."
    );
  }

  /* =====================================================
     5. ENVIAR REQUISIÃ‡ÃƒO AO MERCADO PAGO
     ===================================================== */

  const resposta = await fetch(MERCADO_PAGO_API, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${accessToken}`,
      "Content-Type": "application/json",
      Accept: "application/json",
      "X-Idempotency-Key": dados.chaveIdempotencia,
    },
    body: JSON.stringify({
      type: "online",
      processing_mode: "manual",
      total_amount: dados.valorTotal.toFixed(2),
      external_reference: String(dados.pedidoId),
      items: itens,
    }),
    cache: "no-store",
    signal: AbortSignal.timeout(15000),
  });

  /* =====================================================
     6. VALIDAR RESPOSTA
     ===================================================== */

  if (!resposta.ok) {
    // NÃ£o registrar o corpo da resposta ou credenciais
    // em logs de produÃ§Ã£o.
    throw new Error(
      `Mercado Pago retornou HTTP ${resposta.status}.`
    );
  }

  const resultado =
    (await resposta.json()) as RespostaOrderMercadoPago;

  if (
    typeof resultado.id !== "string" ||
    !resultado.id ||
    typeof resultado.checkout_url !== "string"
  ) {
    throw new Error(
      "Resposta de pagamento incompleta."
    );
  }

  // Verifica se a URL pertence a um domÃ­nio HTTPS
  // esperado do Mercado Pago.
  const urlPagamento = new URL(resultado.checkout_url);

  if (
    urlPagamento.protocol !== "https:" ||
    !(
      urlPagamento.hostname === "mercadopago.com.br" ||
      urlPagamento.hostname.endsWith(".mercadopago.com.br")
    )
  ) {
    throw new Error(
      "URL de pagamento invÃ¡lida."
    );
  }

  /* =====================================================
     7. RETORNAR DADOS PARA O BACKEND
     ===================================================== */

  return {
    mercadoPagoOrderId: resultado.id,
    checkoutUrl: urlPagamento.toString(),
    status: resultado.status ?? "created",
  };
}

import { prisma } from "../../lib/prisma";

/* =========================================================
   CONTROLLER - CRIAR PEDIDO
   ========================================================= */

export async function criarPedidoController(
  request: NextRequest
) {
  try {
    /* =====================================================
       1. IDENTIFICAR O COOKIE DE AUTENTICAÃ‡ÃƒO
       ===================================================== */

    const token = request.cookies.get("padua_session")?.value;

    if (!token) {
      return NextResponse.json(
        { erro: "UsuÃ¡rio nÃ£o autenticado." },
        { status: 401 }
      );
    }

    /* =====================================================
       2. VALIDAR O JWT
       ===================================================== */

    let usuarioId: number;
    let versaoSessao: number;

    try {
      const payload = await validarAuthToken(token);

      usuarioId = Number(payload.usuarioId);
      versaoSessao = Number(payload.versaoSessao);
    } catch {
      return NextResponse.json(
        { erro: "SessÃ£o invÃ¡lida ou expirada." },
        { status: 401 }
      );
    }

    /* =====================================================
       3. VERIFICAR SE A SESSÃƒO FOI REVOGADA
       ===================================================== */

    const usuario = await prisma.usuario.findUnique({
      where: {
        id: usuarioId,
      },
      select: {
        id: true,
        versaoSessao: true,
      },
    });

    if (
      !usuario ||
      usuario.versaoSessao !== versaoSessao
    ) {
      return NextResponse.json(
        { erro: "SessÃ£o revogada. FaÃ§a login novamente." },
        { status: 401 }
      );
    }

    /* =====================================================
       4. VALIDAR A ORIGEM DA REQUISIÃ‡ÃƒO
       =====================================================
       ProteÃ§Ã£o contra requisiÃ§Ãµes externas que tentem
       utilizar o cookie de autenticaÃ§Ã£o do cliente.
       ===================================================== */

    const origin = request.headers.get("origin");

    if (!origin) {
      return NextResponse.json(
        { erro: "Origem da requisiÃ§Ã£o nÃ£o informada." },
        { status: 403 }
      );
    }

    try {
      const origem = new URL(origin);
      const destino = new URL(request.url);

      if (origem.origin !== destino.origin) {
        return NextResponse.json(
          { erro: "Origem da requisiÃ§Ã£o nÃ£o autorizada." },
          { status: 403 }
        );
      }
    } catch {
      return NextResponse.json(
        { erro: "Origem da requisiÃ§Ã£o invÃ¡lida." },
        { status: 403 }
      );
    }

    /* =====================================================
       5. VALIDAR O CORPO DA REQUISIÃ‡ÃƒO
       ===================================================== */

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { erro: "JSON invÃ¡lido." },
        { status: 400 }
      );
    }

    if (
      !body ||
      typeof body !== "object" ||
      Array.isArray(body)
    ) {
      return NextResponse.json(
        { erro: "Dados do pedido invÃ¡lidos." },
        { status: 400 }
      );
    }

    /* =====================================================
       6. PREPARAR A CRIAÃ‡ÃƒO DO PEDIDO
       =====================================================
       O Service serÃ¡ implementado na prÃ³xima etapa.

       Ainda nÃ£o criamos pedidos nem movimentamos estoque.
       ===================================================== */

    return NextResponse.json(
      {
        mensagem:
          "AutenticaÃ§Ã£o validada. A criaÃ§Ã£o de pedidos ainda nÃ£o estÃ¡ habilitada.",
      },
      {
        status: 501,
        headers: {
          "Cache-Control": "no-store",
        },
      }
    );
  } catch (error) {
    console.error(
      "[PEDIDO CONTROLLER] Erro ao processar pedido:",
      error
    );

    return NextResponse.json(
      { erro: "NÃ£o foi possÃ­vel processar o pedido." },
      { status: 500 }
    );
  }
}
